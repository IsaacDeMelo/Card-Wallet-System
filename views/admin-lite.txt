    const cardSearch = document.getElementById('cardSearch');
    const userSearch = document.getElementById('userSearch');
    const cardsList = document.getElementById('cardsList');
    const usersList = document.getElementById('usersList');
    const cardsCount = document.getElementById('cardsCount');
    const usersCount = document.getElementById('usersCount');
    const ownersCount = document.getElementById('ownersCount');
    const ownedCardsCount = document.getElementById('ownedCardsCount');
    const cardsVisible = document.getElementById('cardsVisible');
    const usersVisible = document.getElementById('usersVisible');
    const liteCardId = document.getElementById('liteCardId');
    const liteCategoria = document.getElementById('liteCategoria');
    const liteTitle = document.getElementById('liteTitle');
    const liteCreateBtn = document.getElementById('liteCreateBtn');
    const liteCreateStatus = document.getElementById('liteCreateStatus');
    const userModal = document.getElementById('userModal');
    const closeModalBtn = document.getElementById('closeModalBtn');
    const modalUserName = document.getElementById('modalUserName');
    const modalUserInfo = document.getElementById('modalUserInfo');
    const modalInventoryList = document.getElementById('modalInventoryList');
    const modalCardSelect = document.getElementById('modalCardSelect');
    const modalCardQtd = document.getElementById('modalCardQtd');
    const modalAddCardBtn = document.getElementById('modalAddCardBtn');
    const modalDeleteUserBtn = document.getElementById('modalDeleteUserBtn');
    const modalActionStatus = document.getElementById('modalActionStatus');

    let adminPassword = '';
    let overviewData = { cards: [], users: [] };
    let currentModalUser = null;

    function escapeHtml(v) {
      return String(v ?? '').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;').replace(/'/g,'&#39;');
    }

    function setStatus(msg, type, el = statusText) {
      el.textContent = msg;
      el.className = 'status';
      if (type) el.classList.add(type);
      el.classList.remove('hidden');
      if (el !== statusText) setTimeout(() => el.classList.add('hidden'), 4000);
    }

    function getHeaders() {
      return { 'Content-Type': 'application/json', 'x-admin-password': adminPassword };
    }

    async function fetchOverview() {
      const r = await fetch('/api/admin-lite/overview', { headers: getHeaders() });
      const d = await r.json();
      if (!r.ok || !d.success) throw new Error(d.message || 'Falha ao carregar.');
      return d;
    }

    function countOwnedCards(users) {
      return users.reduce((t, u) => {
        const cards = Array.isArray(u.cards) ? u.cards : [];
        return t + cards.reduce((s, e) => s + (Number(e.quantity) || 0), 0);
      }, 0);
    }

    async function loadData() {
      setStatus('Sincronizando...', '', statusText);
      try {
        const data = await fetchOverview();
        overviewData = { cards: Array.isArray(data.cards) ? data.cards : [], users: Array.isArray(data.users) ? data.users : [] };
        cardsCount.textContent = overviewData.cards.length;
        usersCount.textContent = overviewData.users.length;
        ownersCount.textContent = overviewData.users.filter(u => Array.isArray(u.cards) && u.cards.length > 0).length;
        ownedCardsCount.textContent = countOwnedCards(overviewData.users);
        renderCards(); renderUsers();
        if (currentModalUser) {
          const updated = overviewData.users.find(u => u.username === currentModalUser.username);
          if (updated) openUserModal(updated.username); else closeModal();
        }
        statusText.classList.add('hidden');
      } catch (err) { setStatus(err.message, 'error'); }
    }

    function renderCards() {
      const s = cardSearch.value.trim().toLowerCase();
      const cards = overviewData.cards.filter(c => String(c.title||'').toLowerCase().includes(s) || String(c.cardId||'').toLowerCase().includes(s));
      cardsVisible.textContent = `${cards.length} cartas`;
      if (!cards.length) { cardsList.innerHTML = '<div class="empty-state">Nenhuma carta encontrada.</div>'; return; }
      cardsList.innerHTML = cards.map(card => `
        <div class="item-card">
          <div class="item-head">
            <span class="item-name">${escapeHtml(card.title)}</span>
            <span class="item-id">${escapeHtml(card.cardId)}</span>
          </div>
          <p class="item-meta">Categoria: ${escapeHtml(card.categoria || 'N/A')}</p>
          <button class="btn btn-danger delete-card-btn" data-card-id="${escapeHtml(card.cardId)}" style="font-size:0.78rem;padding:6px 12px;">
            Excluir do BD
          </button>
        </div>`).join('');
    }

    async function deleteCard(cardId) {
      if (!cardId || !window.confirm(`Excluir a carta [${cardId}]? Todos os usuários que a possuem irão perdê-la.`)) return;
      try {
        const r = await fetch(`/api/admin-lite/cards/${encodeURIComponent(cardId)}`, { method: 'DELETE', headers: getHeaders() });
        const d = await r.json();
        if (!r.ok || !d.success) throw new Error(d.message);
        setStatus(d.message || 'Excluída.', 'ok', statusText);
        await loadData();
      } catch (err) { setStatus(err.message, 'error', statusText); }
    }

    liteCreateBtn.addEventListener('click', async () => {
      const cardId = liteCardId.value.trim(), title = liteTitle.value.trim(), categoria = liteCategoria.value.trim();
      if (!cardId || !title || !categoria) { setStatus('Preencha ID, Categoria e Nome.', 'warn', liteCreateStatus); return; }
      setStatus('Forjando...', '', liteCreateStatus);
      try {
        const r = await fetch('/api/admin-lite/cards', { method: 'POST', headers: getHeaders(), body: JSON.stringify({ adminPassword, cardId, title, categoria, draw: '', url: '' }) });
        const d = await r.json();
        if (!r.ok || !d.success) throw new Error(d.message);
        setStatus(`Carta "${title}" forjada!`, 'ok', liteCreateStatus);
        liteCardId.value = ''; liteTitle.value = ''; liteCategoria.value = '';
        await loadData();
      } catch (err) { setStatus(err.message, 'error', liteCreateStatus); }
    });

    function renderUsers() {
      const s = userSearch.value.trim().toLowerCase();
      const users = overviewData.users.filter(u => String(u.username||'').toLowerCase().includes(s) || String(u.clan||'').toLowerCase().includes(s));
      usersVisible.textContent = `${users.length} usuários`;
      if (!users.length) { usersList.innerHTML = '<div class="empty-state">Nenhum shinobi encontrado.</div>'; return; }
      usersList.innerHTML = users.map(user => {
        const cards = Array.isArray(user.cards) ? user.cards : [];
        return `
          <div class="item-card">
            <div class="item-head">
              <span class="item-name">${escapeHtml(user.username)}</span>
              <button class="btn btn-ghost manage-user-btn" data-username="${escapeHtml(user.username)}" style="padding:5px 12px;font-size:0.78rem;border-color:rgba(80,200,120,0.3);color:var(--accent);">
                Gerenciar
              </button>
            </div>
            <p class="item-meta">Zap: ${escapeHtml(user.whatsapp || '—')}</p>
            <div class="user-cards-row">
              ${cards.length ? cards.map(e => `<span class="user-card-tag">${escapeHtml(e.title)} <span>×${escapeHtml(e.quantity)}</span></span>`).join('') : '<span style="font-size:0.75rem;color:var(--muted2);">Bolsa vazia</span>'}
            </div>
          </div>`;
      }).join('');
    }

    function openUserModal(username) {
      const user = overviewData.users.find(u => u.username === username);
      if (!user) return;
      currentModalUser = user;
      modalUserName.textContent = user.username;
      modalUserInfo.textContent = `Clã: ${String(user.clan || '—').toUpperCase()} · Zap: ${user.whatsapp || 'Não informado'}`;
      const cards = Array.isArray(user.cards) ? user.cards : [];
      modalInventoryList.innerHTML = !cards.length
        ? '<div class="empty-state" style="padding:16px;font-size:0.82rem;">Bolsa vazia.</div>'
        : cards.map(e => `
            <div class="inventory-row">
              <div class="inventory-info">
                <span class="inventory-name">${escapeHtml(e.title)}</span>
                <span class="inventory-id">${escapeHtml(e.cardId)}</span>
              </div>
              <div class="inventory-actions">
                <span class="qty-badge">×${escapeHtml(e.quantity)}</span>
                <button class="btn-remove remove-user-card-btn" data-card-id="${escapeHtml(e.cardId)}">−1</button>
              </div>
            </div>`).join('');
      modalCardSelect.innerHTML = overviewData.cards.map(c => `<option value="${escapeHtml(c.cardId)}">${escapeHtml(c.title)} · ${escapeHtml(c.cardId)}</option>`).join('') || '<option value="">Sem cartas</option>';
      modalCardQtd.value = 1;
      modalActionStatus.classList.add('hidden');
      userModal.classList.add('active');
    }

    function closeModal() { userModal.classList.remove('active'); currentModalUser = null; }

    modalAddCardBtn.addEventListener('click', async () => {
      if (!currentModalUser) return;
      const cardId = modalCardSelect.value, quantity = Math.max(1, Number(modalCardQtd.value) || 1);
      if (!cardId) { setStatus('Selecione uma carta.', 'warn', modalActionStatus); return; }
      setStatus('Entregando...', '', modalActionStatus);
      try {
        const r = await fetch('/api/admin/assign', { method: 'POST', headers: getHeaders(), body: JSON.stringify({ adminPassword, username: currentModalUser.username, cardId, quantity }) });
        const d = await r.json();
        if (!r.ok || !d.success) throw new Error(d.message);
        setStatus(`${quantity}× entregues com sucesso.`, 'ok', modalActionStatus);
        await loadData();
      } catch (err) { setStatus(err.message, 'error', modalActionStatus); }
    });

    modalInventoryList.addEventListener('click', async (e) => {
      const btn = e.target.closest('.remove-user-card-btn');
      if (!btn || !currentModalUser) return;
      setStatus('Removendo 1 cópia...', '', modalActionStatus);
      try {
        const r = await fetch('/api/admin/remove', { method: 'POST', headers: getHeaders(), body: JSON.stringify({ adminPassword, username: currentModalUser.username, cardId: btn.getAttribute('data-card-id'), quantity: 1 }) });
        const d = await r.json();
        if (!r.ok || !d.success) throw new Error(d.message);
        setStatus('Removido.', 'ok', modalActionStatus);
        await loadData();
      } catch (err) { setStatus(err.message, 'error', modalActionStatus); }
    });

    modalDeleteUserBtn.addEventListener('click', async () => {
      if (!currentModalUser) return;
      const username = currentModalUser.username;
      if (!window.confirm(`DELETAR o usuário [${username}] permanentemente?`)) return;
      modalDeleteUserBtn.textContent = 'Excluindo...'; modalDeleteUserBtn.disabled = true;
      try {
        const r = await fetch(`/api/admin/users/${encodeURIComponent(username)}`, { method: 'DELETE', headers: getHeaders() });
        const d = await r.json();
        if (!r.ok || !d.success) throw new Error(d.message);
        closeModal();
        setStatus(`Usuário ${username} excluído.`, 'ok', statusText);
        await loadData();
      } catch (err) { setStatus(err.message, 'error', modalActionStatus); }
      finally { modalDeleteUserBtn.textContent = 'Excluir Usuário Permanentemente'; modalDeleteUserBtn.disabled = false; }
    });

    usersList.addEventListener('click', e => { const btn = e.target.closest('.manage-user-btn'); if (btn) openUserModal(btn.getAttribute('data-username')); });
    cardsList.addEventListener('click', e => { const btn = e.target.closest('.delete-card-btn'); if (btn) deleteCard(btn.getAttribute('data-card-id')); });
    closeModalBtn.addEventListener('click', closeModal);
    userModal.addEventListener('click', e => { if (e.target === userModal) closeModal(); });
    cardSearch.addEventListener('input', renderCards);
    userSearch.addEventListener('input', renderUsers);

    function unlock() { authBox.classList.add('hidden'); content.classList.remove('hidden'); }

    unlockBtn.addEventListener('click', async () => {
      adminPassword = passwordInput.value.trim();
      if (!adminPassword) return setStatus('Informe a senha.', 'warn');
      try { await loadData(); unlock(); sessionStorage.setItem('adminLitePassword', adminPassword); }
      catch (err) { adminPassword = ''; setStatus(err.message, 'error'); }
    });

    passwordInput.addEventListener('keydown', e => { if (e.key === 'Enter') unlockBtn.click(); });
    refreshBtn.addEventListener('click', () => loadData());
    goWalletBtn.addEventListener('click', () => window.location.href = '/wallet');
    logoutBtn.addEventListener('click', () => { sessionStorage.removeItem('adminLitePassword'); window.location.reload(); });

    window.addEventListener('load', () => {
      const saved = sessionStorage.getItem('adminLitePassword');
      if (saved) { adminPassword = saved; loadData().then(unlock).catch(() => sessionStorage.removeItem('adminLitePassword')); }
    });
  </script>
</body>
</html>
